### pipeline.py

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import mlflow
import mlflow.sklearn
import logging

# Setup logging
logging.basicConfig(level=logging.INFO)

def load_data(file_path):
    """Load the dataset from a CSV file."""
    logging.info("Loading data from %s", file_path)
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        logging.error("Error loading data: %s", str(e))
        raise

def preprocess_data(data):
    """Preprocess the data by splitting into features and target."""
    logging.info("Preprocessing data")
    try:
        X = data.drop('Class', axis=1)
        y = data['Class']
        return X, y
    except Exception as e:
        logging.error("Error preprocessing data: %s", str(e))
        raise

def train_model(X_train, y_train, random_state=42):
    """Train a RandomForest model."""
    logging.info("Training model")
    try:
        model = RandomForestClassifier(random_state=random_state)
        model.fit(X_train, y_train)
        return model
    except Exception as e:
        logging.error("Error training model: %s", str(e))
        raise

def evaluate_model(model, X_test, y_test):
    """Evaluate the model's performance."""
    logging.info("Evaluating model")
    try:
        y_pred = model.predict(X_test)
        accuracy = accuracy_score(y_test, y_pred)
        logging.info("Model accuracy: %.2f", accuracy)
        return accuracy
    except Exception as e:
        logging.error("Error evaluating model: %s", str(e))
        raise

def run_pipeline(file_path, random_state=42):
    """Run the entire pipeline."""
    try:
        data = load_data(file_path)
        X, y = preprocess_data(data)
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=random_state)
        model = train_model(X_train, y_train, random_state=random_state)
        accuracy = evaluate_model(model, X_test, y_test)
        return accuracy
    except Exception as e:
        logging.error("Error running pipeline: %s", str(e))
        raise
```
