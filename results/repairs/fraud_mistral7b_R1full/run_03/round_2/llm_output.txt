from sklearn.model_selection import train_test_split

import mlflow
import mlflow.sklearn
import numpy as np

def load_data(file_path):

def preprocess_data(data):
    # Add any necessary preprocessing steps like scaling, encoding, etc.
    # For simplicity, I'll just return the data as it is.
    return data, data['Amount'].astype(np.float64)

def train_model(X_train, y_train):
    with mlflow.start_run():
        model = RandomForestClassifier(random_state=42)  # Seed the random state for reproducibility
        model.fit(X_train, y_train)
        mlflow.sklearn.log_model(model, "random-forest")

    run_id = mlflow.get_run_id()
    print(f"Trained model saved with run ID: {run_id}")
