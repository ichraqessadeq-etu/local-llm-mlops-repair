import mlflow
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.utils.multiclass import type_of_target

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    return (data.drop('Amount', axis=1), data['Amount'])

def train_model(X_train, y_train):
    with mlflow.start_run():
        model = RandomForestClassifier(random_state=42)
        model.fit(X_train, y_train)
        mlflow.pyfunc.log_model(model, 'model')
        mlflow.log_artifact(artifact_path='model', artifact_rel_path='models')
        mlflow.log_metric('training_loss', model.loss_)
        mlflow.log_metric('training_score', model.score_)

        # Check if the target y is of a non-regression type
        y_type = type_of_target(y_train)
        if y_type not in [
            "binary",
            "multiclass",
            "multiclass-multioutput",
            "multilabel-indicator",
            "multilabel-sequences",
        ]:
            raise ValueError("Unknown label type: %r" % y_type)
