```python
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_param, log_artifact, start_run, log_metric
import numpy as np

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    # Preprocessing steps like filling NaN, encoding categorical variables, scaling features, etc.
    # For simplicity, I'm assuming the data is clean and doesn't require preprocessing.
    data = data.copy()
    data.rename(columns={'Features': 'feature'}, inplace=True)
    return data['feature'], data['Congestion']

def train_model(X_train, y_train):
    with start_run(run_name="RandomForest_Congestion_Detector"):
        try:
            model = RandomForestClassifier(random_state=42)
            model.fit(X_train, y_train)

            log_param("learning_rate", model.get_params()['max_depth'])
            log_artifact("model.pkl", (model))

            return model
        except Exception as e:
            log_metric("training_error", 1)
            print(f"An error occurred during training: {e}")
```
