1. pipeline.py:

```python
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_param, log_artifact, start_run, end_run, register_model, log_metric
import pytest

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    # Preprocessing steps like filling NA values, encoding categorical variables, etc.
    # For simplicity, I'm assuming data is already preprocessed.
    return data['Features'], data['Congestion']

def train_model(X_train, y_train, random_state=42):
    model = RandomForestClassifier(random_state=random_state)
    model.fit(X_train, y_train)
    return model

def train(file_path):
    data = load_data(file_path)
    X, y = preprocess_data(data)
    with start_run():
        model = train_model(X_train=X, y_train=y)
        log_artifact(model, "model")
        log_metric("train_loss", model.loss_)
        log_param("num_trees", model.n_estimators)
        log_param("random_state", random_state)
        register_model(model, "random_forest_congestion_detector")
```

2. test_pipeline.py:

```python
import unittest
import numpy as np
from pipeline import load_data, preprocess_data, train_model

class TestPipeline(unittest.TestCase):
    def setUp(self):
        self.data = load_data("data/traffic.csv")

    def test_load_data(self):
        self.assertIsInstance(self.data, pd.DataFrame)

    def test_preprocess_data(self):
        X, y = preprocess_data(self.data)
        self.assertIsInstance(X, pd.DataFrame)
        self.assertIsInstance(y, pd.Series)

    def test_train_model(self):
        X, y = preprocess_data(self.data)
        model = train_model(X_train=X, y_train=y)
        self.assertIsInstance(model, RandomForestClassifier)

if __name__ == "__main__":
    unittest.main()
```

In the updated pipeline.py, I added a random_state parameter to the train\_model function to resolve the reproducibility violation. In test\_pipeline.py, I added a setUp method to preload the data and a test\_train\_model function to test the trained model.
