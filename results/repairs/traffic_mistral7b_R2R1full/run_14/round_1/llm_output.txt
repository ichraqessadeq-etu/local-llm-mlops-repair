 ```python
# pipeline.py
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_param, log_artifact, start_run, end_run, register_model
from mlflow.sklearn import save_model

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    data = data[['temp', 'rain_1h', 'snow_1h', 'clouds_all', 'weather_main', 'IsHoliday', 'Hour', 'DayOfWeek', 'Month']]
    return (data, data['Congestion'])

def train_model(X_train, y_train):
    model = RandomForestClassifier(random_state=42)
    model.fit(X_train, y_train)
    return model
```

# test_pipeline.py
from sklearn.ensemble import RandomForestClassifier
import pandas as pd
import unittest
from pipeline import load_data, preprocess_data, train_model

class TestPipeline(unittest.TestCase):
    def setUp(self):
        self.file_path = 'data/traffic.csv'

    def test_load_data(self):
        data = load_data(self.file_path)
        self.assertIsInstance(data, pd.DataFrame)

    def test_preprocess_data(self):
        data = load_data(self.file_path)
        X, y = preprocess_data(data)
        self.assertIsInstance(X, pd.DataFrame)
        self.assertIsInstance(y, pd.Series)

    def test_train_model(self):
        data = load_data(self.file_path)
        X, y = preprocess_data(data)
        model = train_model(X, y)
        self.assertIsInstance(model, RandomForestClassifier)

if __name__ == '__main__':
    unittest.main()
