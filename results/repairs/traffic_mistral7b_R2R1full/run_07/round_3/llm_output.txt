 To fix the issue, we need to remove the 'clouds_all' column from the dataset before training the model, as it contains string values that cannot be converted to float.

Here's the updated `pipeline.py` file:

```python
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
import mlflow
import mlflow.sklearn

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    features = ['temp', 'rain_1h', 'snow_1h', 'weather_main', 'IsHoliday', 'Hour', 'DayOfWeek', 'Month']
    data[features] = data[features].astype(float)
    data.drop('clouds_all', axis=1, inplace=True)
    return (data[features], data['Congestion'])

def train_model(X_train, y_train):
    with mlflow.start_run():
        model = RandomForestClassifier(random_state=42)
        model.fit(X_train, y_train)
        mlflow.sklearn.log_model(model, 'random-forest')
```
