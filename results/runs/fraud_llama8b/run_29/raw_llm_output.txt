Here is the code for the production-ready CI/CD pipeline:

**pipeline.py**
```python
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import mlflow.sklearn
import mlflow

def load_data(file_path):
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        print(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    try:
        X = data.drop(['Class'], axis=1)
        y = data['Class']
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X)
        X_train, X_test, y_train, y_test = train_test_split(X_scaled, y, test_size=0.2, random_state=42)
        return X_train, X_test, y_train, y_test
    except Exception as e:
        print(f"Error preprocessing data: {e}")
        raise

def train_model(X_train, y_train):
    try:
        with mlflow.start_run():
            mlflow.log_param("model_type", "RandomForestClassifier")
            mlflow.log_param("n_estimators", 100)
            mlflow.log_param("max_depth", 5)
            mlflow.log_param("min_samples_split", 2)
            mlflow.log_param("min_samples_leaf", 1)
            model = RandomForestClassifier(n_estimators=100, max_depth=5, min_samples_split=2, min_samples_leaf=1)
            model.fit(X_train, y_train)
            mlflow.sklearn.log_model(model, "model")
            return model
    except Exception as e:
        print(f"Error training model: {e}")
        raise
```

**test_pipeline.py**
```python
import pytest
from pipeline import load_data, preprocess_data, train_model

def test_load_data():
    data = load_data('data/fraud.csv')
    assert data is not None

def test_preprocess_data():
    data = load_data('data/fraud.csv')
    X_train, X_test, y_train, y_test = preprocess_data(data)
    assert X_train is not None
    assert X_test is not None
    assert y_train is not None
    assert y_test is not None

def test_train_model():
    data = load_data('data/fraud.csv')
    X_train, X_test, y_train, y_test = preprocess_data(data)
    model = train_model(X_train, y_train)
    assert model is not None
```

**.github/workflows/ci_cd.yml**
```yml
name: CI/CD

on:
  push:
    branches:
      - main

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: |
          pip install -r requirements.txt

      - name: Run tests
        run: |
          pytest

      - name: Train model
        run: |
          python pipeline.py train_model

      - name: Log model to MLflow
        uses: mlflow/mlflow-action@v1
        with:
          mlflow-tracking-uri: 'https://example.com/mlflow'

      - name: Deploy model
        uses: actions/deploy@v1
        with:
          deployment-name: 'fraud-detector'
          deployment-type: 'docker'
          deployment-image: 'fraud-detector:latest'
```

**requirements.txt**
```
pandas==1.2.0
scikit-learn==1.0.2
mlflow==1.20.0
```

Note that you'll need to replace `https://example.com/mlflow` with your actual MLflow tracking URI.

Also, make sure to add the `pipeline.py` and `test_pipeline.py` files to your repository and update the `requirements.txt` file to include the necessary dependencies.

This pipeline assumes that you have a `data/fraud.csv` file in your repository, which contains the data for the fraud detection system. The pipeline loads the data, preprocesses it, trains a random forest classifier, logs the model to MLflow, and deploys it as a Docker image. The tests in `test_pipeline.py` ensure that the pipeline functions correctly.
