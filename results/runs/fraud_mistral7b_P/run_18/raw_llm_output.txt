3. GitHub Actions YAML:

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2
      - name: Set up Python 3.7
        uses: actions/setup-python@v2
        with:
          python-version: 3.7
      - name: Install dependencies
        run: |
          pip install mlflow scikit-learn pytest
      - name: Test and Train
        run: |
          pytest test_pipeline.py
          mlflow run --python pipeline.py --run-name my_run --track-runs
```

This example uses GitHub Actions to trigger the pipeline on every push or pull request. The pipeline checks out the code, sets up Python 3.7, installs the required dependencies, runs the tests, and trains the model using MLflow.
