1. pipeline.py

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_artifact, pyfunc, track

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    return data.drop(columns=['Class']), data['Class']

def train_model(X_train, y_train):
    with mlflow.start_run():
        model = RandomForestClassifier()
        model.fit(X_train, y_train)
        mlflow.sklearn.log_model(model, "model")
        track(f"Training Score: {model.score(X_train, y_train)}")
        return model
```

2. test_pipeline.py

```python
import unittest
from pipeline import load_data, preprocess_data, train_model
from sklearn.metrics import accuracy_score

class TestPipeline(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.data = load_data("data/fraud.csv")
        cls.X, cls.y = preprocess_data(cls.data)

    def test_train_model(self):
        X_train, X_test, y_train, y_test = train_test_split(self.X, self.y, test_size=0.3, random_state=42)
        model = train_model(X_train, y_train)
        predictions = model.predict(X_test)
        self.assertGreaterEqual(accuracy_score(y_test, predictions), 0.8)

if __name__ == "__main__":
    unittest.main()
```

3. GitHub Actions YAML

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python environment
        uses: actions/setup-python@v2
        with:
          python-version: 3.7

      - name: Install dependencies
        run: |
          pip install -r requirements.txt
          pip install mlflow==1.30.1 scikit-learn==1.0.2

      - name: Test
        run: pytest test_pipeline.py

      - name: Train and log model
        run: python pipeline.py --tracking-uri "mlflow_tracking_uri"

      - name: Upload artifacts
        uses: mlflow/mlflow-github-action@v1
        with:
          tracking-uri: "mlflow_tracking_uri"
          artifact-path: "target/mlflow"
```
