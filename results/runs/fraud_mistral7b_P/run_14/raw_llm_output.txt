1. pipeline.py

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_artifact, pyfunc, track

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    return data.drop(columns=['Class']), data['Class']

def train_model(X_train, y_train):
    model = RandomForestClassifier()
    with mlflow.start_run():
        model.fit(X_train, y_train)
        mlflow.sklearn.log_model(model, "model")
        mlflow.log_artifact(artifact=model, artifact_path="models")
        pyfunc.log_model(model, artifact_path="models")
        track(f"Training Score: {model.score(X_train, y_train)}")
```

3. GitHub Actions YAML

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python 3.7
      uses: actions/setup-python@v2
      with:
        python-version: 3.7

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        python -m pip install scikit-learn==1.0.2 mlflow==1.30.1

    - name: Test and Train
      run: |
        python -m pytest test_pipeline.py
        python pipeline.py
```
