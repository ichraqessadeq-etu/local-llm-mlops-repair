3. GitHub Actions YAML:

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Setup Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install dependencies
      run: |
        pip install -r requirements.txt

    - name: Test
      run: pytest test_pipeline.py

    - name: Train and Log Model
      run: python pipeline.py

    - name: Upload Artifacts
      uses: actions/upload-artifact@v2
      with:
        name: models
        path: .mlflow/artifacts/models
```

Make sure to create a `requirements.txt` file to list all the dependencies needed for your project.
