3. GitHub Actions YAML

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up MLflow
      uses: mlflow/mlflow-docker@v1
      with:
        mlflow_version: 1.21.1
        python: 3.8

    - name: Test and Train
      run: |
        python test_pipeline.py
        python pipeline.py data/fraud.csv

    - name: Push MLflow artifacts
      uses: mlflow/mlflow-docker@v1
      with:
        mlflow_version: 1.21.1
        python: 3.8
      env:
        MLFLOW_TRACKING_URI: http://<your-mlflow-tracking-server>:5000
      run: |
        mlflow models logs -t random-forest -P model_version=1
```
