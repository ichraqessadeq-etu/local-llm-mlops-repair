3. GitHub Actions YAML:

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Setup Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install dependencies
      run: |
        pip install pandas mlflow pytest

    - name: Test
      run: pytest test_pipeline.py

    - name: Train and Log Model
      run: python pipeline.py --file data/fraud.csv

  deploy:
    needs: build
    runs-on: ubuntu-latest

    steps:
    - name: Save model artifact
      run: mlflow artifact log --path ./ --tag model_version $(mlflow run --python pipeline.py --file data/fraud.csv | grep "Model with version" | awk '{print $NF}')
```
