Here's a production-ready CI/CD pipeline for a RandomForest fraud detection system using Python, MLflow, and GitHub Actions. This pipeline includes data loading, preprocessing, model training, and error handling.

### pipeline.py

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import mlflow
import mlflow.sklearn
import logging

logging.basicConfig(level=logging.INFO)

def load_data(file_path):
    """Load data from a CSV file."""
    try:
        data = pd.read_csv(file_path)
        logging.info("Data loaded successfully.")
        return data
    except FileNotFoundError:
        logging.error("File not found.")
        raise
    except pd.errors.ParserError:
        logging.error("Error parsing CSV file.")
        raise

def preprocess_data(data):
    """Preprocess the data by splitting into features and target."""
    try:
        X = data.drop('Class', axis=1)
        y = data['Class']
        logging.info("Data preprocessed successfully.")
        return X, y
    except KeyError:
        logging.error("Target column 'Class' not found.")
        raise

def train_model(X_train, y_train):
    """Train a RandomForest classifier."""
    try:
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        logging.info("Model trained successfully.")
        return model
    except Exception as e:
        logging.error(f"Error training model: {e}")
        raise

def save_model(model, experiment_name, run_name):
    """Save the trained model to MLflow."""
    try:
        mlflow.set_experiment(experiment_name)
        with mlflow.start_run(run_name=run_name):
            mlflow.sklearn.log_model(model, "model")
            logging.info("Model saved to MLflow successfully.")
    except Exception as e:
        logging.error(f"Error saving model to MLflow: {e}")
        raise
```

### test_pipeline.py

```python
import pytest
from pipeline import load_data, preprocess_data, train_model, save_model

@pytest.fixture
def data():
    """Fixture to load and preprocess data."""
    data_path = 'data/fraud.csv'
    data = load_data(data_path)
    X, y = preprocess_data(data)
    return X, y

def test_load_data():
    """Test loading data."""
    with pytest.raises(FileNotFoundError):
        load_data('nonexistent_file.csv')

def test_preprocess_data():
    """Test preprocessing data."""
    with pytest.raises(KeyError):
        preprocess_data(pd.DataFrame())

def test_train_model():
    """Test training the model."""
    X, y = data()
    model = train_model(X, y)
    assert isinstance(model, RandomForestClassifier)

def test_save_model():
    """Test saving the model."""
    X, y = data()
    model = train_model(X, y)
    save_model(model, 'fraud_detection', 'test_run')
```

### GitHub Actions YAML

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.7

    - name: Install dependencies
      run: |
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest

    - name: Build and push to MLflow
      uses: mlflow/mlflow-github-actions@v1
      with:
        experiment-name: fraud_detection
        run-name: main_run
        mlflow-server: https://mlflow.example.com
        mlflow-token: ${{ secrets.MLFLOW_TOKEN }}
        mlflow-username: ${{ secrets.MLFLOW_USERNAME }}
        mlflow-password: ${{ secrets.MLFLOW_PASSWORD }}
```

### Notes

- **MLflow**: The pipeline uses MLflow for model tracking and versioning. Ensure you have an MLflow server running and configure the `mlflow-server`, `mlflow-token`, `mlflow-username`, and `mlflow-password` in the GitHub Actions YAML.
- **Error Handling**: The pipeline includes basic error handling to catch and log exceptions during data loading, preprocessing, model training, and saving.
- **Environment**: The pipeline is designed to run on Python 3.7 with the specified versions of MLflow and scikit-learn.
