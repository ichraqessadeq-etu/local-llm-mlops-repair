Here is the code for the pipeline, test file, and GitHub Actions YAML:

**pipeline.py**
```python
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import mlflow.sklearn
import mlflow

def load_data(file_path):
    """Load air quality data from CSV file"""
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        mlflow.log_error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    """Preprocess data by selecting features and encoding categorical variables"""
    try:
        # Select features
        X = data[['PT08.S1(CO)', 'PT08.S2(NMHC)', 'PT08.S3(NOx)', 'PT08.S4(NO2)', 'PT08.S5(O3)', 'NOx(GT)', 'C6H6(GT)', 'T', 'RH', 'AH', 'Hour', 'DayOfWeek', 'Month']]

        # Encode categorical variables
        X['Hour'] = pd.Categorical(X['Hour']).codes
        X['DayOfWeek'] = pd.Categorical(X['DayOfWeek']).codes
        X['Month'] = pd.Categorical(X['Month']).codes

        # Scale data
        scaler = StandardScaler()
        X = scaler.fit_transform(X)

        # Split data into features and target
        y = data['Anomaly']

        return X, y
    except Exception as e:
        mlflow.log_error(f"Error preprocessing data: {e}")
        raise

def train_model(X_train, y_train):
    """Train a RandomForest classifier on the preprocessed data"""
    try:
        # Train model
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)

        # Log model to MLflow
        mlflow.sklearn.log_model(model, "random_forest")

        return model
    except Exception as e:
        mlflow.log_error(f"Error training model: {e}")
        raise
```

**test_pipeline.py**
```python
import pytest
from pipeline import load_data, preprocess_data, train_model

def test_load_data():
    data = load_data('data/air_quality.csv')
    assert data is not None

def test_preprocess_data():
    data = load_data('data/air_quality.csv')
    X, y = preprocess_data(data)
    assert X is not None
    assert y is not None

def test_train_model():
    data = load_data('data/air_quality.csv')
    X, y = preprocess_data(data)
    model = train_model(X, y)
    assert model is not None
```

**.github/workflows/ci_cd.yml**
```yml
name: CI/CD

on:
  push:
    branches:
      - main

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python environment
        uses: actions/setup-python@v2
        with:
          python-version: '3.7'

      - name: Install dependencies
        run: |
          pip install -r requirements.txt

      - name: Run tests
        run: |
          pytest test_pipeline.py

      - name: Train model
        run: |
          python pipeline.py

      - name: Log model to MLflow
        run: |
          mlflow.sklearn.log_model("random_forest")

      - name: Deploy model
        uses: appleboy/ssh-action@v0.4.0
        with:
          host: ${{ secrets.HOST }}
          username: ${{ secrets.USERNAME }}
          password: ${{ secrets.PASSWORD }}
          script: |
            mkdir -p /path/to/model
            cp /path/to/model/random_forest.pkl /path/to/model/
```

Note that you'll need to replace the `secrets` placeholders in the GitHub Actions YAML with your actual host, username, and password.

Also, make sure to create a `requirements.txt` file in the root of your repository with the following contents:
```
mlflow==1.30.1
scikit-learn==1.0.2
```
This will ensure that the correct versions of MLflow and scikit-learn are installed during the CI/CD pipeline.
