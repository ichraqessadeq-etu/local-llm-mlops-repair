3. **GitHub Actions YAML**

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    name: Build and Test
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Setup Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install dependencies
      run: |
        pip install pandas numpy scikit-learn mlflow pytest

    - name: Test
      run: |
        pytest test_pipeline.py

  train:
    name: Train Model
    needs: build
    runs-on: ubuntu-latest

    steps:
    - name: Load Data
      run: |
        DATA_FILE=data/traffic.csv
        DATA=$(python pipeline.py load_data $DATA_FILE)

    - name: Preprocess Data
      run: |
        X, y = pipeline.preprocess_data($DATA)

    - name: Train Model
      run: |
        python pipeline.py train_model $X $y

    - name: Log Model
      run: |
        mlflow login --host $MLFLOW_TRACKING_URI --username $MLFLOW_USERNAME --password $MLFLOW_PASSWORD
        mlflow push artifact . -t congestion-detector
```

This pipeline includes a build job that runs tests, and a train job that loads the data, preprocesses it, trains the model, and logs it using MLflow. The MLflow tracking URI, username, and password should be set as secrets in your GitHub repository settings.
