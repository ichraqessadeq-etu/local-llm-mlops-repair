3. GitHub Actions YAML

```yaml
name: CI/CD for Traffic Management System

on: [push, pull_request]

jobs:
  build:
    name: Build and Test
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Setup Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install Dependencies
      run: |
        pip install -r requirements.txt

    - name: Test
      run: pytest test_pipeline.py

    - name: Train Model
      run: python pipeline.py

    - name: Upload Model
      if: needs:build.succeeded()
      uses: mlflow/mlflow-action@v2
      with:
        track: dev
        artifact: model
        mlflow-version: v1.21.0
```
