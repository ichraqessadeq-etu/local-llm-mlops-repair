3. GitHub Actions YAML

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Setup Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install dependencies
      run: |
        pip install pandas mlflow scikit-learn randomforest-py

    - name: Lint and Test
      run: |
        pip install pytest
        pytest test_pipeline.py

    - name: Train Model
      run: python pipeline.py

    - name: Upload Model
      uses: mlflow/mlflow-action@v2
      with:
        track: dev
        artifact_path: .
```
