from sklearn.ensemble import RandomForestClassifier
import mlflow
import mlflow.sklearn

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    # Preprocessing steps like filling NaN, encoding categorical variables, etc.
    # For simplicity, I'm assuming that the data is already preprocessed
    return data['Features'], data['Congestion']

def train_model(X_train, y_train):
    with mlflow.start_run():
        model = RandomForestClassifier()
        model.fit(X_train, y_train)
        mlflow.sklearn.log_model(model, "random_forest")

import unittest
from pipeline import load_data, preprocess_data, train_model

class TestPipeline(unittest.TestCase):
    def test_load_data(self):
        data = load_data('data/traffic.csv')
        self.assertIsInstance(data, pd.DataFrame)

    def test_preprocess_data(self):
        data = load_data('data/traffic.csv')
        X, y = preprocess_data(data)
        self.assertIsInstance(X, pd.DataFrame)
        self.assertIsInstance(y, pd.Series)

    def test_train_model(self):
        data = load_data('data/traffic.csv')
        X, y = preprocess_data(data)
        model = train_model(X, y)
        self.assertIsInstance(model, RandomForestClassifier)

if __name__ == "__main__":
    unittest.main()

3. GitHub Actions YAML:

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Setup Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install dependencies
      run: |
        pip install pandas mlflow sklearn

    - name: Test pipeline
      run: |
        python test_pipeline.py

    - name: Train model
      run: |
        python pipeline.py
        mlflow login --host $MLFLOW_TRACKING_URI --username $MLFLOW_USERNAME --password $MLFLOW_PASSWORD
        mlflow push artifact . -f model
```

This pipeline uses GitHub Actions to automate the build, test, and deployment of your model. It checks out your code, sets up Python, installs dependencies, runs tests, trains the model, and pushes the trained model to MLflow. You'll need to replace `$MLFLOW_TRACKING_URI`, `$MLFLOW_USERNAME`, and `$MLFLOW_PASSWORD` with your actual MLflow tracking URI, username, and password.
