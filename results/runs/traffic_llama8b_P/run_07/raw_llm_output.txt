Here is the code for the MLOps pipeline:

**pipeline.py**
```python
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import mlflow.sklearn
import mlflow

def load_data(file_path):
    """Load data from CSV file"""
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        mlflow.log_error(f"Error loading data: {str(e)}")
        raise

def preprocess_data(data):
    """Preprocess data by scaling and encoding categorical variables"""
    try:
        # Scale numerical features
        scaler = StandardScaler()
        data[['temp', 'rain_1h', 'snow_1h', 'clouds_all']] = scaler.fit_transform(data[['temp', 'rain_1h', 'snow_1h', 'clouds_all']])

        # Encode categorical variables
        data['weather_main'] = pd.Categorical(data['weather_main']).codes
        data['IsHoliday'] = pd.Categorical(data['IsHoliday']).codes

        # Split data into features (X) and target (y)
        X = data.drop('Congestion', axis=1)
        y = data['Congestion']

        return X, y
    except Exception as e:
        mlflow.log_error(f"Error preprocessing data: {str(e)}")
        raise

def train_model(X_train, y_train):
    """Train a RandomForest model on the preprocessed data"""
    try:
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        mlflow.sklearn.log_model(model, "random_forest_model")
        return model
    except Exception as e:
        mlflow.log_error(f"Error training model: {str(e)}")
        raise
```

**test_pipeline.py**
```python
import pytest
from pipeline import load_data, preprocess_data, train_model

def test_load_data():
    data = load_data('data/traffic.csv')
    assert data is not None

def test_preprocess_data():
    data = load_data('data/traffic.csv')
    X, y = preprocess_data(data)
    assert X is not None
    assert y is not None

def test_train_model():
    data = load_data('data/traffic.csv')
    X, y = preprocess_data(data)
    model = train_model(X, y)
    assert model is not None
```

**.github/workflows/ci_cd.yml**
```yml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python environment
        uses: actions/setup-python@v2
        with:
          python-version: '3.7'

      - name: Install dependencies
        run: |
          pip install -r requirements.txt

      - name: Train model
        run: |
          python pipeline.py

      - name: Test pipeline
        run: |
          pytest test_pipeline.py

      - name: Log model to MLflow
        run: |
          mlflow.sklearn.log_model("random_forest_model", "models")

      - name: Deploy to production
        run: |
          # Deploy to production environment (e.g. AWS, GCP, etc.)
```

Note that this is just one possible implementation of a CI/CD pipeline for an MLOps project. You may need to modify the pipeline to fit your specific use case and requirements.
