**pipeline.py**
```python
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import mlflow.sklearn
import mlflow

def load_data(file_path):
    """Loads data from CSV file"""
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        mlflow.log_error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    """Preprocesses data by encoding categorical variables and scaling numerical variables"""
    try:
        # Encode categorical variables
        data['weather_main'] = pd.Categorical(data['weather_main']).codes
        data['Month'] = pd.Categorical(data['Month']).codes
        data['DayOfWeek'] = pd.Categorical(data['DayOfWeek']).codes

        # Scale numerical variables
        scaler = StandardScaler()
        data[['temp', 'rain_1h', 'snow_1h', 'clouds_all']] = scaler.fit_transform(data[['temp', 'rain_1h', 'snow_1h', 'clouds_all']])

        # Split data into features (X) and target (y)
        X = data.drop('Congestion', axis=1)
        y = data['Congestion']

        return X, y
    except Exception as e:
        mlflow.log_error(f"Error preprocessing data: {e}")
        raise

def train_model(X_train, y_train):
    """Trains a RandomForest model"""
    try:
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        mlflow.sklearn.log_model(model, "random_forest_model")
        return model
    except Exception as e:
        mlflow.log_error(f"Error training model: {e}")
        raise
```

**test_pipeline.py**
```python
import pandas as pd
import pytest
from pipeline import load_data, preprocess_data, train_model

def test_load_data():
    data = load_data('data/traffic.csv')
    assert data.shape[0] > 0

def test_preprocess_data():
    data = load_data('data/traffic.csv')
    X, y = preprocess_data(data)
    assert X.shape[0] > 0
    assert y.shape[0] > 0

def test_train_model():
    data = load_data('data/traffic.csv')
    X, y = preprocess_data(data)
    model = train_model(X, y)
    assert isinstance(model, RandomForestClassifier)
```

**.github/workflows/ci_cd.yml**
```yml
name: CI/CD

on:
  push:
    branches:
      - main

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.7'

      - name: Install dependencies
        run: |
          pip install mlflow scikit-learn

      - name: Run tests
        run: |
          pytest test_pipeline.py

      - name: Train and deploy model
        run: |
          python pipeline.py
          mlflow run . -P model_name="random_forest_model"
```

Note: This YAML file assumes that you have a `main` branch in your repository and that you want to run the CI/CD pipeline on push events to that branch. You can modify the `on` section to suit your needs.

Also, make sure to create a `MLFLOW_TRACKING_URI` environment variable in your GitHub Actions workflow to point to your MLflow tracking server. You can do this by adding the following step to your YAML file:
```yml
- name: Set up MLflow
  run: |
    export MLFLOW_TRACKING_URI="https://your-tracking-server.com"
```
