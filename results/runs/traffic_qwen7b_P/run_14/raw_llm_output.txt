### 1. Python Module `pipeline.py`

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from sklearn.exceptions import NotFittedError
import mlflow
import mlflow.sklearn
import mlflow.pyfunc

def load_data(file_path):
    """Load data from a CSV file."""
    return pd.read_csv(file_path)

def preprocess_data(data):
    """Preprocess data for model training."""
    # Assuming 'Congestion' is the target column
    X = data.drop(columns=['Congestion'])
    y = data['Congestion']
    return X, y

def train_model(X_train, y_train):
    """Train a RandomForestClassifier model."""
    with mlflow.start_run():
        mlflow.log_param("model", "RandomForestClassifier")
        mlflow.log_param("features", X_train.columns.tolist())
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        y_pred = model.predict(X_train)
        accuracy = accuracy_score(y_train, y_pred)
        mlflow.log_metric("train_accuracy", accuracy)
        mlflow.sklearn.log_model(model, "model")
        return model

def predict(model, X):
    """Predict congestion using the trained model."""
    if not isinstance(model, RandomForestClassifier):
        raise NotFittedError("Model is not fitted")
    return model.predict(X)
```

### 2. Pytest File `test_pipeline.py`

```python
import pandas as pd
from pipeline import load_data, preprocess_data, train_model, predict

def test_load_data():
    data = load_data('data/traffic.csv')
    assert not data.empty
    assert 'Congestion' in data.columns

def test_preprocess_data():
    data = load_data('data/traffic.csv')
    X, y = preprocess_data(data)
    assert not X.empty
    assert not y.empty
    assert len(X) == len(y)

def test_train_model():
    data = load_data('data/traffic.csv')
    X, y = preprocess_data(data)
    model = train_model(X, y)
    assert isinstance(model, RandomForestClassifier)
    assert model.is_fitted_

def test_predict():
    data = load_data('data/traffic.csv')
    X, y = preprocess_data(data)
    model = train_model(X, y)
    X_test, _ = preprocess_data(data.sample(frac=0.2, random_state=42))
    predictions = predict(model, X_test)
    assert isinstance(predictions, pd.Series)
    assert len(predictions) == len(X_test)
```

### 3. GitHub Actions YAML

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python 3.7
        uses: actions/setup-python@v2
        with:
          python-version: 3.7

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run tests
        run: pytest

      - name: Build MLflow model
        run: |
          mlflow models build -m pipeline.py --name traffic-congestion-detector --version 1

      - name: Push MLflow model to registry
        run: |
          mlflow models serve --model-uri models:/traffic-congestion-detector/1 --port 5000
```
