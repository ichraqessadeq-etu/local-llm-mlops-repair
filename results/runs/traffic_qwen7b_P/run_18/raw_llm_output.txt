### 1. `pipeline.py`

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from sklearn.preprocessing import LabelEncoder
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder
import mlflow
import mlflow.sklearn
from mlflow.models.signature import infer_signature

def load_data(file_path):
    """Load data from a CSV file."""
    return pd.read_csv(file_path)

def preprocess_data(data):
    """Preprocess the data for model training."""
    # Separate features and target
    X = data.drop(columns=['Congestion'])
    y = data['Congestion']
    
    # Categorical features
    categorical_features = ['weather_main', 'IsHoliday']
    
    # Numerical features
    numerical_features = ['temp', 'rain_1h', 'snow_1h', 'clouds_all', 'Hour', 'DayOfWeek', 'Month']
    
    # Create transformers
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', SimpleImputer(strategy='median'), numerical_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ])
    
    # Apply preprocessing
    X_preprocessed = preprocessor.fit_transform(X)
    
    return X_preprocessed, y

def train_model(X_train, y_train):
    """Train a RandomForestClassifier and log it using MLflow."""
    with mlflow.start_run():
        # Define the model
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        
        # Train the model
        model.fit(X_train, y_train)
        
        # Log the model
        mlflow.sklearn.log_model(model, "model")
        
        # Log the parameters
        mlflow.log_param("n_estimators", 100)
        mlflow.log_param("random_state", 42)
        
        # Log the metrics
        y_pred = model.predict(X_train)
        accuracy = accuracy_score(y_train, y_pred)
        mlflow.log_metric("accuracy", accuracy)
        
        # Infer the signature
        signature = infer_signature(X_train, y_pred)
        
        # Save the signature
        mlflow.sklearn.save_model(model, "model", signature=signature)
        
        return model
```

### 3. `.github/workflows/ci_cd.yml`

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python 3.7
        uses: actions/setup-python@v2
        with:
          python-version: 3.7

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run tests
        run: pytest test_pipeline.py

      - name: Push MLflow model to registry
        run: |
          mlflow models serve --model-uri runs:/$(mlflow search runs --max-results 1 --order-by "start_time desc" --output-format json | jq -r '.[0].run_id')/model --no-conda -p 5000 &
          sleep 10
          curl -X POST http://localhost:5000/invocations -H "Content-Type: application/json" -d '{"data": [[20, 0, 0, 0.5, "Clear", False, 12, 1, 1]]}'
        env:
          MLFLOW_TRACKING_URI: https://mlflow.example.com
```
