### 1. Python Module `pipeline.py`

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from mlflow import log_param, log_metric, log_model

def load_data(file_path):
    """Load data from a CSV file."""
    return pd.read_csv(file_path)

def preprocess_data(data):
    """Preprocess data by separating features and target."""
    X = data.drop('Congestion', axis=1)
    y = data['Congestion']
    return X, y

def train_model(X_train, y_train):
    """Train a RandomForestClassifier and log metrics using MLflow."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    # Log model parameters
    log_param('n_estimators', model.n_estimators)
    log_param('random_state', model.random_state)
    
    # Log model metrics
    y_pred = model.predict(X_train)
    accuracy = accuracy_score(y_train, y_pred)
    log_metric('accuracy', accuracy)
    
    # Log the model
    log_model(model, "random_forest_model")
    
    return model
```

### 3. GitHub Actions YAML

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python 3.7
      uses: actions/setup-python@v2
      with:
        python-version: 3.7

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: pytest test_pipeline.py

    - name: Push MLflow model
      run: |
        mlflow models serve -m mlruns/0/$(mlflow search experiments --max-results 1 --order-by "last_updated desc" --output-format json | jq -r '.experiments[0].experiment_id')/artifacts/random_forest_model --port 5000
      env:
        MLFLOW_TRACKING_URI: http://mlflow-server:5000
```
