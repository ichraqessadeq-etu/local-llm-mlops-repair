### 1. Python Module: `pipeline.py`

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from sklearn.preprocessing import StandardScaler
import mlflow
import mlflow.sklearn

def load_data(file_path):
    """Load data from a CSV file."""
    return pd.read_csv(file_path)

def preprocess_data(data):
    """Preprocess the data by separating features and target."""
    X = data.drop(columns=['Anomaly'])
    y = data['Anomaly']
    return X, y

def train_model(X_train, y_train):
    """Train a RandomForestClassifier and log it using MLflow."""
    with mlflow.start_run():
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        mlflow.log_param("n_estimators", 100)
        mlflow.sklearn.log_model(model, "model")
        return model

def evaluate_model(model, X_test, y_test):
    """Evaluate the model and log the accuracy."""
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    mlflow.log_metric("accuracy", accuracy)
    return accuracy
```

```yaml
on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

    runs-on: ubuntu-latest
    steps:
      - name: Checkout code

      - name: Set up Python 3.7

        with:
          python-version: '3.7'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run tests
        run: pytest test_pipeline.py

      - name: Train model and log to MLflow
        run: |
          python pipeline.py
        env:
          MLFLOW_TRACKING_URI: https://mlflow.yourdomain.com

```
