1. pipeline.py:

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_artifact, track, pyfunc, start_run, end_run

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    return data, data['Congestion']

def train_model(X_train, y_train):
    with start_run():
        model = RandomForestClassifier()
        model.fit(X_train, y_train)
        pyfunc.log_model(model, "model")
        track(model.feature_importances_)
    return model
```

2. test_pipeline.py:

```python
import unittest
from unittest.mock import patch
from pipeline import load_data, preprocess_data, train_model

class TestPipeline(unittest.TestCase):
    @patch('pipeline.pd.read_csv')
    def test_load_data(self, mock_read_csv):
        mock_data = {'features': [1, 2, 3], 'Congestion': [0, 1, 0]}
        mock_read_csv.return_value = mock_data
        result = load_data('data/traffic.csv')
        expected = mock_data
        self.assertEqual(result, expected)

    @patch('pipeline.pd.DataFrame')
    def test_preprocess_data(self, mock_dataframe):
        mock_data = {'features': [1, 2, 3], 'Congestion': [0, 1, 0]}
        mock_dataframe.return_value = mock_data
        result, _ = preprocess_data(mock_data)
        expected_X = mock_data[['temp', 'rain_1h', 'snow_1h', 'clouds_all', 'weather_main', 'IsHoliday', 'Hour', 'DayOfWeek', 'Month']]
        expected_y = mock_data['Congestion']
        self.assertEqual(result[0], expected_X)
        self.assertEqual(result[1], expected_y)

if __name__ == '__main__':
    unittest.main()
```

3. GitHub Actions YAML:

```yaml
name: CI/CD Pipeline
on: [push, pull_request]
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2
      - name: Set up Python 3.7
        uses: actions/setup-python@v2
        with:
          python-version: 3.7
      - name: Install dependencies
        run: |
          pip install -r requirements.txt
      - name: Test
        run: pytest test_pipeline.py
      - name: Train and log model
        run: python pipeline.py --tracking-uri=mlflow_tracking_uri --artifact-path=./artifacts
      - name: Upload artifacts
        uses: actions/upload-artifact@v2
        with:
          name: model
          path: ./artifacts
```

In this example, replace `mlflow_tracking_uri` with your MLflow tracking server URL. Also, make sure to create a `requirements.txt` file containing the necessary dependencies:

```
mlflow==1.30.1
scikit-learn==1.0.2
pandas==1.3.5
pytest==6.2.5
```
