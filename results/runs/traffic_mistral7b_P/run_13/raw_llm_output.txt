**1. pipeline.py**

```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_artifact, pyfunc, track

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    data = data.dropna()
    return np.array(data.drop('Congestion', axis=1)), data['Congestion']

def train_model(X_train, y_train):
    with mlflow.start_run():
        model = RandomForestClassifier()
        model.fit(X_train, y_train)
        pyfunc.log_model(model, "model", registered_model_name="congestion_detector")
        track(f"Congestion Detector Training Loss", model.loss_)
        track(f"Congestion Detector Training Score", model.score(X_train, y_train))
        return model
```

**3. GitHub Actions YAML**

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2

      - name: Set up Python 3.7
        uses: actions/setup-python@v2
        with:
          python-version: 3.7

      - name: Install dependencies
        run: |
          pip install -r requirements.txt
          pip install mlflow==1.30.1 scikit-learn==1.0.2

      - name: Train and Test Model
        run: python pipeline.py

      - name: Test
        run: pytest test_pipeline.py

      - name: Upload Artifacts
        uses: mlflow/mlflow-github-action@v1
        with:
          track_runs: true
          artifact_paths:
            - pipeline/model
```
