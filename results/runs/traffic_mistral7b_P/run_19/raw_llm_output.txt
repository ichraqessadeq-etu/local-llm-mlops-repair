import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from mlflow import log_artifact, pyfunc, tracking

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    return data, data['Congestion']

def train_model(X_train, y_train):
    with mlflow.start_run(run_name="RandomForestCongestionDetector"):
        mlflow.log_artifact(file_path="data/traffic.csv")

        model = RandomForestClassifier()
        model.fit(X_train, y_train)
        pyfunc.log_model(model, "model")

        print(f"Model training completed with MLflow run ID: {mlflow.get_run_id()}")
```

2. `test_pipeline.py`:

import unittest
from pipeline import load_data, preprocess_data, train_model

class TestPipeline(unittest.TestCase):
    def setUp(self):
        self.data = load_data("data/traffic.csv")
        self.X, self.y = preprocess_data(self.data)

    def test_train_model(self):
        train_model(self.X, self.y)

if __name__ == "__main__":
    unittest.main()
```

3. GitHub Actions YAML:

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2
      - name: Set up Python 3.7
        uses: actions/setup-python@v2
        with:
          python-version: 3.7
      - name: Install dependencies
        run: |
          pip install mlflow scikit-learn pytest
      - name: Test and Train
        run: |
          pytest test_pipeline.py
          python pipeline.py
```

This pipeline uses MLflow to track the model and data artifacts, pytest for testing, and GitHub Actions for continuous integration and deployment. The target environment is Python 3.7 with MLflow 1.30.1 and scikit-learn 1.0.2.
